    <!-- JAVASCRIPT -->
    <script>
        // ---- State ----
        var cartCount = 2;      // starting count as shown in dashboard
        var cartTotal = 0;

        // ---- Sidebar Navigation ----
        function showPage(pageId, element) {
            document.querySelectorAll('.page-section').forEach(function(page) {
                page.classList.remove('active');
            });
            var target = document.getElementById(pageId);
            if (target) target.classList.add('active');

            document.querySelectorAll('.nav-item').forEach(function(item) {
                item.classList.remove('active');
            });
            if (element) element.classList.add('active');
        }

        // ---- Add to Cart ----
        function addToCart(itemName, price, btn) {
            cartCount = cartCount + 1;
            cartTotal = cartTotal + price;

            // Update dashboard counter
            var cartEl = document.getElementById('cart-count');
            if (cartEl) cartEl.textContent = cartCount;

            // Toast notification
            var toast = document.getElementById('toast');
            var toastMsg = document.getElementById('toast-msg');
            toastMsg.textContent = itemName + " added! (Cart: " + cartCount + " items • ₹" + cartTotal + ")";
            toast.classList.add('show');
            setTimeout(function() { toast.classList.remove('show'); }, 2200);

            // Button feedback
            if (btn) {
                var originalText = btn.innerHTML;
                btn.innerHTML = '<i class="fa-solid fa-check"></i> Added';
                btn.classList.add('added');
                btn.disabled = true;
                setTimeout(function() {
                    btn.innerHTML = originalText;
                    btn.classList.remove('added');
                    btn.disabled = false;
                }, 1500);
            }
        }
    </script>
